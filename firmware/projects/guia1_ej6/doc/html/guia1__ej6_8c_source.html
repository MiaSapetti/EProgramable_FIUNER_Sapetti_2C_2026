132int8_t convertToBcdArray(uint32_t data, uint8_t digits, uint8_t *bcd_number)
133{
134 // Bucle para i desde cantidadDigitos-1 hasta 0
135 for (int8_t i = digits - 1; i >= 0; i--) // Guarda datos en el orden correcto
136 {
137 uint8_t digito = data % 10; // digito = numero MOD 10
138 bcd_number[i] = digito; // puntero[i] = digito
139 data = data / 10; // numero = numero DIV 10
140 }
141 return 0;
142}

147void setBCDtoGPIO(uint8_t bcd_digit, gpioConf_t *gpio_vector)
148{
149 for (uint8_t b = 0; b < 4; b++)
150 {
151 // Se extrae el estado del bit 'b' usando desplazamiento y enmascarando el bit menos significativo
152 if ((bcd_digit >> b) & 0x01)
153 {
154 GPIOOn(gpio_vector[b].pin); // Poner en ALTO ('1')
155 }
156 else
157 {
158 GPIOOff(gpio_vector[b].pin); // Poner en BAJO ('0')
159 }
160 }
161}
