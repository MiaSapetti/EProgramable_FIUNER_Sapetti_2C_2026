166void FuncUART(void* param){
167 uint8_t tecla;
168 UartReadByte(UART_PC, &tecla);
169 UartSendByte(UART_PC, (char *) &tecla);
170
171 if(tecla == 'O' || tecla == 'o'){
172 on = !on;
173 }
174 else if(tecla == 'H' || tecla == 'h'){
175 hold = !hold;
176 }
177 else if(tecla == 'I' || tecla == 'i'){
178 unidades_cm = !unidades_cm;
179 }
180 else if(tecla == 'M' || tecla == 'm'){
181 modo_maximo = !modo_maximo;
182 }
183 else if(tecla == 'F' || tecla == 'f'){
184 if(periodo_medir_us > PERIOD_MIN_MEDIR_US){
185 periodo_medir_us -= STEP_PERIOD_US;
186 TimerUpdatePeriod(TIMER_A, periodo_medir_us);
187 UartSendString(UART_PC, "Velocidad aumentada\r\n");
188 }
189 }
190 else if(tecla == 'S' || tecla == 's'){
191 if(periodo_medir_us < PERIOD_MAX_MEDIR_US){
192 periodo_medir_us += STEP_PERIOD_US;
193 TimerUpdatePeriod(TIMER_A, periodo_medir_us);
194 UartSendString(UART_PC, "Velocidad disminuida\r\n");
195 }
196 }
197}

210static void MostrarTask(void *pvParameter){
211 LcdItsE0803Init();
212
213 while(true){
214 ulTaskNotifyTake(pdTRUE, portMAX_DELAY);
215 if(on){
216 // 1. CONTROL DE LEDS (Tiempo real)
217 if (distancia < 10) {
218 LedOff(LED_1);
219 LedOff(LED_2);
220 LedOff(LED_3);
221 } else if (distancia >= 10 && distancia < 20) {
222 LedOn(LED_1);
223 LedOff(LED_2);
224 LedOff(LED_3);
225 } else if (distancia >= 20 && distancia <= 30) {
226 LedOn(LED_1);
227 LedOn(LED_2);
228 LedOff(LED_3);
229 } else if (distancia > 30) {
230 LedOn(LED_1);
231 LedOn(LED_2);
232 LedOn(LED_3);
233 }
234
235 // 2. CONTROL DEL DISPLAY BCD Y VARIABLE A MOSTRAR
236 if(modo_maximo){
237 distancia_lcd = max_distancia;
238 }
239 else if(!hold){
240 distancia_lcd = distancia;
241 }
242
243 LcdItsE0803Write(distancia_lcd);
244
245 // 3. ENVÍO VÍA PUERTO SERIE
246 if (modo_maximo) {
247 UartSendString(UART_PC, "MAX: ");
248 }
249
250 UartSendString(UART_PC, (char *)UartItoa(distancia_lcd, 10));
251
252 if(unidades_cm){
253 UartSendString(UART_PC, " cm\r\n");
254 }
255 else {
256 UartSendString(UART_PC, " inch\r\n");
257 }
258 }
259 else {
260 LcdItsE0803Off();
261 LedsOffAll();
262 }
263 }

357
