102static void MostrarTask(void *pvParameter){
103 // Inicialización del LCD dentro de la tarea
104 LcdItsE0803Init();
105
106 while(true){
107 if(on){
108 // 1. CONTROL DE LEDS (Tiempo real)
109 if (distancia < 10) {
110 LedOff(LED_1);
111 LedOff(LED_2);
112 LedOff(LED_3);
113 } else if (distancia >= 10 && distancia < 20) {
114 LedOn(LED_1);
115 LedOff(LED_2);
116 LedOff(LED_3);
117 } else if (distancia >= 20 && distancia <= 30) {
118 LedOn(LED_1);
119 LedOn(LED_2);
120 LedOff(LED_3);
121 } else if (distancia > 30) {
122 LedOn(LED_1);
123 LedOn(LED_2);
124 LedOn(LED_3);
125 }
126
127 // 2. CONTROL DEL DISPLAY BCD (Retiene valor si 'hold' es true)
128 if(!hold){
129 distancia_lcd = distancia;
130 }
131
132 // Muestra directamente el entero (0 a 999) en el display
133 LcdItsE0803Write(distancia_lcd);
134
135 }
136 else {
137 // Si 'on' es false, apaga el display y los LEDs
138 LcdItsE0803Off();
139 LedsOffAll();
140 }
141
142 vTaskDelay(REFRESCO_MEDICION_MS / portTICK_PERIOD_MS);
143 }
144}

178static void TeclasTask(void *pvParameter){

190 // TEC2: Alterna HOLD
