207static void EnviarResultadoFinal(void)
208{
209 // Deja terminar de enviar la última ventana de ProcesamientoTask
210 vTaskDelay(1500 / portTICK_PERIOD_MS);
211
212 if (n_ventanas == 0) {
213 ESP_LOGW(TAG, "Sin ventanas validas para el resultado final");
214 return;
215 }
216
217 float rms_total = sqrtf(rms2_acum / (float)n_ventanas);
218
219 // Espectro promedio y su pico
220 float espectro_prom[BIN_ENVIO_MAX + 1];
221 int idx_pico = 1;
222 float max_prom = 0.0f;
223 for (int i = 1; i <= BIN_ENVIO_MAX; i++) {
224 espectro_prom[i] = espectro_acum[i] / (float)n_ventanas;
225 if (espectro_prom[i] > max_prom) {
226 max_prom = espectro_prom[i];
227 idx_pico = i;
228 }
229 }
230 float f_total = f[idx_pico];
231
232 // Envío: espectro promedio y métricas finales
233 char msg[32];
234 EnviarLinea("*GC*");
235 for (int i = 1; i <= BIN_ENVIO_MAX; i++) {
236 snprintf(msg, sizeof(msg), "*GX%.1fY%.2f*", f[i], espectro_prom[i]);
237 EnviarLinea(msg);
238 }
239 if (rms_total > UMBRAL_RMS_TEMBLOR) {
240 snprintf(msg, sizeof(msg), "*D%.2f*", f_total);
241 }
242 else {
243 snprintf(msg, sizeof(msg), "*DSin temblor*");
244 }
245 EnviarLinea(msg);
246 snprintf(msg, sizeof(msg), "*S%.2f*", rms_total);
247 EnviarLinea(msg);
248
249 ESP_LOGI(TAG, "RESULTADO: RMS=%.2f m/s^2 | Frec=%.2f Hz | %d ventanas", rms_total, f_total, n_ventanas);
250}

256static void AdquisicionTask(void *pvParameter){
257 float ax_g, ay_g, az_g;
258 while(true){
259 ulTaskNotifyTake(pdTRUE, portMAX_DELAY);
260 // Solo adquiere cuando hay una prueba activa
261 if(estado_actual != MODO_REPOSO){
262 // 1. Obtener mediciones de los 3 ejes
263 ax_g = ReadXValue();
264 ay_g = ReadYValue();
265 az_g = ReadZValue();
266
267 // 2. Convertir d g a m/s²
268 float ax_ms2 = ax_g * GRAVEDAD_MS2;
269 float ay_ms2 = ay_g * GRAVEDAD_MS2;
270 float az_ms2 = az_g * GRAVEDAD_MS2;
271
272 // 3. Calcular el módulo vectorial
273 modulo_ms2 = sqrtf((ax_ms2 * ax_ms2) + (ay_ms2 * ay_ms2) + (az_ms2 * az_ms2));
274
275 // 4. Guardar en el buffer
276 //buffer_aceleracion[buffer_index] = az_ms2; // cambio de modulo_ms2 (pruebas)
277 buffer_aceleracion[buffer_index] = modulo_ms2;
278 buffer_index++;
279
280 // 5. Verificar si el buffer de muestras está lleno para ser procesado
281 if (buffer_index >= BUFFER_SIZE) {
282 buffer_index = 0; // Reiniciar índice
283
284 // 6. Cargar datos en el buffer de procesamiento con una copia desacoplada de seguridad
285 memcpy(buffer_procesamiento, buffer_aceleracion, sizeof(buffer_aceleracion));
286
287 // 7. Avisar a la Tarea de Procesamiento que hay un nuevo bloque listo
288 if (Procesamiento_task_handle != NULL) {
289 xTaskNotifyGive(Procesamiento_task_handle);
290 }
291 }
292 }
293 }

300static void ProcesamientoTask(void *pvParameter){
301 float fft_input[BUFFER_SIZE];
302 while(true){
303 // Espera bloqueada hasta que la Tarea de Adquisición complete las 128 muestras
304 ulTaskNotifyTake(pdTRUE, portMAX_DELAY);
305
306 // 1. Calcular el valor medio (componente continua DC / Gravedad)
307 float suma = 0.0f;
308 for (int i = 0; i < BUFFER_SIZE; i++) {
309 suma += buffer_procesamiento[i];
310 }
311 float media_dc = suma / (float)BUFFER_SIZE;
312
313 // 2. Remoción de DC (centrar señal en 0) y cálculo de RMS
314 for (int i = 0; i < BUFFER_SIZE; i++) {
315 fft_input[i] = buffer_procesamiento[i] - media_dc;
316 }
317
318 // 3. Filtrado IIR utilizando las funciones de la cátedra
319 HiPassFilter(fft_input, buffer_filt, BUFFER_SIZE);
320 LowPassFilter(buffer_filt, buffer_filt2, BUFFER_SIZE);
321
322 // RMS de la señal sin DC (sin filtrar): debe dar ~9 m/s² para todos los casos de igual amplitud
323 //float suma_cuad_raw = 0.0f;
324 //for (int i = 0; i < BUFFER_SIZE; i++) {
325 // float x = buffer_procesamiento[i] - media_dc;
326 // suma_cuad_raw += x * x;
327 //}
328 //float rms_raw = sqrtf(suma_cuad_raw / (float)BUFFER_SIZE);
329
330 // 4. Cálculo de RMS (Métrica de severidad sobre el buffer filtrado)
331 float suma_cuadrados = 0.0f;
332 for (int i = 0; i < BUFFER_SIZE; i++) {
333 suma_cuadrados += (buffer_filt2[i] * buffer_filt2[i]);
334 }
335 rms_temblor = sqrtf(suma_cuadrados / (float)BUFFER_SIZE);
336
337 // 5. Cálculo de FFT utilizando la biblioteca de la cátedra
338 FFTFrequency(SAMPLE_FREC, BUFFER_SIZE, f); //vector eje de frecuencias
339 FFTMagnitude(buffer_filt2, espectro_filt_fft, BUFFER_SIZE); //espectro de la señal ya filtrada (por alto y por bajo)
340
341 // 6. Búsqueda de la Frecuencia Dominante en el espectro filtrado
342 float max_magnitud = 0.0f;
343 uint16_t idx_max = 0;
344
345 // Evaluar desde el bin 1 (~0.78 Hz) hasta N/2 (50 Hz, límite Nyquist)
346 for (int16_t i = 1; i < (BUFFER_SIZE / 2); i++) {
347 if (espectro_filt_fft[i] > max_magnitud) {
348 max_magnitud = espectro_filt_fft[i];
349 idx_max = i;
350 }
351 }
352 frecuencia_dominante = f[idx_max];
353
354 // Calculo de métricas acumuladas y envío de resultados por UART y BLE
355 if (primera_ventana) {
356 primera_ventana = false; // descarta el transitorio
357 } else {
358 for (int i = 0; i < BUFFER_SIZE / 2; i++) espectro_acum[i] += espectro_filt_fft[i];
359 rms2_acum += rms_temblor * rms_temblor;
360 n_ventanas++;
361 }
362 // 7. Transmisión del espectro por UART y Bluetooth SPP (frecuencia y magnitud)
363 // Envío en formato para Serial Plotter: >voltaje:VALOR\r\n
364 char msg[32];
365 // Borra el espectro anterior
366 EnviarLinea("*GC*");
367 for (int i = 1; i <= BIN_ENVIO_MAX; i++) {
368 snprintf(msg, sizeof(msg), "*GX%.1fY%.2f*", f[i], espectro_filt_fft[i]);
369 EnviarLinea(msg);
370 }
371
372 // 8. Transmisión de métricas finales (RMS y Frecuencia Dominante) para la App Movil
373 snprintf(msg, sizeof(msg), "*F%.2f*", frecuencia_dominante);
374 EnviarLinea(msg);
375 snprintf(msg, sizeof(msg), "*R%.2f*", rms_temblor);
376 EnviarLinea(msg);
377
378 // 9. Transmisión limpia por UART exclusivamente para Serial Oscilloscope (PC)
379 for (int i = 1; i <= BIN_ENVIO_MAX; i++) {
380 char msg_pc[32];
381 snprintf(msg_pc, sizeof(msg_pc), ">magnitud:%.2f\r\n", espectro_filt_fft[i]);
382 UartSendString(UART_PC, msg_pc);
383 vTaskDelay(5 / portTICK_PERIOD_MS); // Pequeña pausa para no saturar el buffer UART
384 }
385 }

391static void PruebasTask(void *pvParameter){
392 while(true){
393 ulTaskNotifyTake(pdTRUE, portMAX_DELAY);
394
395 if(estado_actual == MODO_ESTATICO){
396 ReiniciarAcumuladores();
397 ESP_LOGI(TAG, "--- PRUEBA ESTATICA INICIADA ---");
398
399 // Tono corto indicando inicio
400 BuzzerOn();
401 vTaskDelay(200 / portTICK_PERIOD_MS);
402 BuzzerOff();
403
404 // Esperar el tiempo de la prueba mientras se evalúa el temblor
405 vTaskDelay(TIEMPO_PRUEBA_ESTATICA_MS / portTICK_PERIOD_MS);
406
407 estado_actual = MODO_REPOSO;
408 EnviarResultadoFinal();
409
410 // Tono largo indicando fin de la prueba
411 BuzzerOn();
412 vTaskDelay(1000 / portTICK_PERIOD_MS);
413 BuzzerOff();
414
415 ESP_LOGI(TAG, "--- PRUEBA ESTATICA FINALIZADA ---");
416 }
417 else if(estado_actual == MODO_DINAMICO){
418 ReiniciarAcumuladores();
419 ESP_LOGI(TAG, "--- PRUEBA DINAMICA INICIADA ---");
420
421 // Tono corto indicando inicio
422 BuzzerOn();
423 vTaskDelay(200 / portTICK_PERIOD_MS);
424 BuzzerOff();
425
426 // Espera de tiempo fijo para la realización del movimiento dinámico
427 vTaskDelay(10000 / portTICK_PERIOD_MS);
428
429 estado_actual = MODO_REPOSO;
430 EnviarResultadoFinal();
431
432 // Tono continuo indicando finalización de la prueba dinámica
433 BuzzerOn();
434 vTaskDelay(1000 / portTICK_PERIOD_MS);
435 BuzzerOff();
436
437 ESP_LOGI(TAG, "--- PRUEBA DINAMICA FINALIZADA ---");
438 }

445void app_main(void){
446 // 1. Inicialización de la UART
447 serial_config_t my_uart = {
448 .port = UART_PC,
449 .baud_rate = 921600,
450 .func_p = NULL,
451 .param_p = NULL
452 };
453 UartInit(&my_uart);
454
455 ESP_LOGI(TAG, "Inicializando periféricos...");
456
457 // Inicializaciones de hardware y librerías de la cátedra
458 LedsInit();
459 SwitchesInit();
460 BuzzerInit(GPIO_20); //Pin asignado al Buzzer
461 FFTInit();
462 LowPassInit(SAMPLE_FREC, 15, ORDER_2); // Pasa bajas en 15 Hz (corta ruido térmico/movimiento rápido)
463 HiPassInit(SAMPLE_FREC, 1, ORDER_2); // Pasa altas en 1 Hz (elimina deriva lenta)
464
465 // Inicialización del módulo Bluetooth SPP
466 ble_config_t my_ble = {
467 .device_name = "EVALUADOR_TEMBLOR_BLE",
468 .func_p = NULL
469 };
470 BleInit(&my_ble);
471
472 // Inicializo los Switches de selección de prueba con interrupciones
473 SwitchActivInt(SWITCH_1, &Tecla1_ISR, NULL);
474 SwitchActivInt(SWITCH_2, &Tecla2_ISR, NULL);
475
476
477 // Inicialización del acelerómetro analógico ADXL335
478 if (ADXL335Init()) {
479 ESP_LOGI(TAG, "ADXL335 inicializado con éxito.");
480 } else {
481 ESP_LOGE(TAG, "Error al inicializar ADXL335.");
482 }
483
484 // 3. Configuración del Timer A para el muestreo
485 timer_config_t timer_Adquisicion = {
486 .timer = TIMER_A,
487 .period = CONFIG_BLINK_PERIOD_Adquisicion_US,
488 .func_p = FuncTimerA,
489 .param_p = NULL
490 };
491 TimerInit(&timer_Adquisicion);
492
493 // 4. Creación de las tareas de FreeRTOS
494 xTaskCreate(&AdquisicionTask, "Adquisicion", 4096, NULL, 5, &Adquisicion_task_handle);
495 xTaskCreate(&ProcesamientoTask, "Procesamiento", 4096, NULL, 5, &Procesamiento_task_handle);
496 xTaskCreate(&PruebasTask, "Pruebas", 4096, NULL, 5, &Pruebas_task_handle);
497
498 // 5. Inicio del Timer A para la adquisición de datos
499 TimerStart(timer_Adquisicion.timer);
500
501 while (1) {
502 vTaskDelay(1000 / portTICK_PERIOD_MS);
503 }
504}
